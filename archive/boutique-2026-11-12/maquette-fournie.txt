  const price = Math.floor((normal * (100 - discount)) / 100);

  const used = new Set();
  const cards = Array.from({ length: n }, (_, i) => {
    const rarity = weightedPick(Object.entries({ common: 57, uncommon: 24, rare: 12, epic: 5, legendary: 2 }).map(([value, w]) => ({ value, w })));
    let eligible = CATALOGUE.filter((c) => c.rarity === rarity && !used.has(c.id));
    if (!eligible.length) eligible = CATALOGUE.filter((c) => !used.has(c.id));
    const card = choose(eligible);
    used.add(card.id);
    const finish = weightedPick([{ value: { type: "normal", mult: 1 }, w: 81 }, { value: { type: "glossy", mult: 2.3 }, w: 15 }, { value: { type: "alternative", mult: 3.2 }, w: 4 }]);
    const fullArt = Math.random() < 0.08;
    const base = 3 * PRICE_REF[card.rarity] * finish.mult * (fullArt ? 1.9 : 1);
    const variation = 0.8 + (Math.random() + Math.random()) * 0.2;
    return { ...card, offerId: i, finish: finish.type, fullArt, price: round10(base * 2.75 * variation), sold: false };
  });
  return { daily: { pack, qty, discount, normal, price, sold: false }, cards };
}

// Lucky Boost : chaque palier de rareté, finition et illustration est présent à 70 %,
// les pourcentages absents sont redistribués proportionnellement (docs/boutique-hebdomadaire.md).
function redistribute(base) {
  const all = Object.keys(base);
  let keys = all.filter(() => Math.random() < 0.7);
  if (!keys.length) keys = [all[0]];
  const total = keys.reduce((t, k) => t + base[k], 0);
  const out = Object.fromEntries(all.map((k) => [k, 0]));
  keys.forEach((k) => { out[k] = Math.round((base[k] * 100) / total); });
  out[keys[0]] += 100 - keys.reduce((t, k) => t + out[k], 0);
  return out;
}
const generateLuckyRates = () => ({ rarity: redistribute(DAILY_CARD_RATES.rarity), finish: redistribute(DEFAULT_FINISH), illustration: redistribute(DEFAULT_ART) });

/* Carrousel une offre à la fois, repris de ShopCarousel (kabomane/bizave-tcg) :
   défilement natif à crans, flèches, points, clavier. */
function Carousel({ count, index, onIndex, label, children }) {
  const track = useRef(null), timer = useRef(), current = useRef(index);
  current.current = index;
  const multiple = count > 1;
  useEffect(() => {
    const el = track.current;
    if (!el) return;
    const left = index * el.clientWidth;
    if (Math.abs(el.scrollLeft - left) > 1) el.scrollTo({ left, behavior: "smooth" });
  }, [index, count]);
  useEffect(() => {
    const el = track.current;
    if (!el || typeof ResizeObserver === "undefined") return;
    const observer = new ResizeObserver(() => { el.scrollLeft = current.current * el.clientWidth; });
    observer.observe(el);
    return () => observer.disconnect();
  }, []);
  useEffect(() => () => clearTimeout(timer.current), []);
  const clamp = (n) => Math.max(0, Math.min(count - 1, n));
  const onScroll = () => {
    clearTimeout(timer.current);
    timer.current = setTimeout(() => {
      const el = track.current;
      if (!el || !el.clientWidth) return;
      const n = clamp(Math.round(el.scrollLeft / el.clientWidth));
      if (n !== current.current) onIndex(n);
    }, 120);
  };
  const onKey = (e) => {
    if (!multiple || !["ArrowLeft", "ArrowRight", "Home", "End"].includes(e.key)) return;
    e.preventDefault();
    onIndex(e.key === "Home" ? 0 : e.key === "End" ? count - 1 : clamp(index + (e.key === "ArrowRight" ? 1 : -1)));
  };
  return (
    <div className="carousel" role={multiple ? "region" : undefined} aria-roledescription={multiple ? "carrousel" : undefined} aria-label={multiple ? label : undefined}>
      <div ref={track} className="carousel-track" tabIndex={multiple ? 0 : -1} onScroll={onScroll} onKeyDown={onKey}>
        {React.Children.map(children, (child, i) => <div className="carousel-slide" aria-hidden={i !== index}>{child}</div>)}
      </div>
      {multiple && <>
        <button type="button" className="carousel-prev" aria-label="Offre précédente" disabled={index === 0} onClick={() => onIndex(index - 1)}><Icon name="chevronLeft" /></button>
        <button type="button" className="carousel-next" aria-label="Offre suivante" disabled={index === count - 1} onClick={() => onIndex(index + 1)}><Icon name="chevronRight" /></button>
        <div className="carousel-dots" aria-hidden="true">{Array.from({ length: count }, (_, i) => <span key={i} className={i === index ? "is-current" : undefined} />)}</div>
        <span className="sr-only" role="status">{index + 1} sur {count}</span>
      </>}
    </div>
  );
}

function parisClock() {
  const p = Object.fromEntries(new Intl.DateTimeFormat("en-GB", { timeZone: "Europe/Paris", weekday: "short", hour: "2-digit", minute: "2-digit", second: "2-digit", hour12: false })
    .formatToParts(new Date()).filter((t) => t.type !== "literal").map((t) => [t.type, t.value]));
  const secs = (23 - Number(p.hour)) * 3600 + (59 - Number(p.minute)) * 60 + (60 - Number(p.second));
  const hh = String(Math.max(0, Math.floor(secs / 3600))).padStart(2, "0");
  const mm = String(Math.max(0, Math.floor((secs % 3600) / 60))).padStart(2, "0");
  const day = ["Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"].indexOf(p.weekday);
  const days = day === -1 ? 0 : (8 - day) % 7;
  return { daily: `${hh} h ${mm} min`, weekly: `${days === 0 ? 7 : days} j ${hh} h` };
}

                        </button>
                      </div>
                    </div>
                  </div>
                </section>
                {!noCards && (
                  <section className="card-zone">
                    <div className="section-line">
                      <h3>Cartes du jour · {offers(dailyCards.length)} · <span className="stock-label">{stock(dailyShown)}</span></h3>
                    </div>
                    <CardStage cards={dailyCards} index={Math.min(dailyIndex, dailyCards.length - 1)} onIndex={setDailyIndex} balance={balance} onBuy={buyDailyCard} label="Choisir une carte du jour" soldLabel="Achetée" />
                  </section>
                )}
              </div>
          </section>

          <section className="shop-section" ref={sections.weekly}>
              <div className="shop-section-head"><h2>Cette semaine</h2><span className="section-timer"><Icon name="clock" /><span>{clock.weekly}</span></span></div>
              <div className="weekly-layout">
                <section className="weekly-lucky">
                  <div className="section-line"><h3>Lucky Boost · {luckyRemaining} restant{luckyRemaining > 1 ? "s" : ""}</h3></div>
                  <div className="offer-card">
                    <div className="lucky-preview">
                      <div className="lucky-pack"><Pack p={LUCKY_PACK} /></div>
                      <div className="lucky-details">
                        <h3>Tirage personnalisé</h3>
                        <p>Un booster aux probabilités et au pool uniques, renouvelé chaque lundi.</p>
                        <div className="price-block"><span className="price-value">120</span><span className="price-units">fragments</span></div>
                        <button className="primary-button" disabled={luckyRemaining <= 0 || balance < 120} onClick={buyLucky}>
                          {luckyRemaining <= 0 ? "Épuisé" : <>Acheter <Icon name="arrow" /></>}
                        </button>
                      </div>
                    </div>
                  </div>
                </section>
                <section className="weekly-showcase">
                  <div className="section-line">
                    <h3>Vitrine de la semaine · {offers(weeklyCards.length)} · <span className="stock-label">{stock(weeklyShown)}</span></h3>
                  </div>
                  <CardStage cards={weeklyCards} index={weeklyIndex} onIndex={setWeeklyIndex} balance={balance} onBuy={buyWeeklyCard} label="Choisir une carte de la vitrine" soldLabel="Épuisée" />
                </section>
              </div>
              <section className="section-block">
                <div className="section-line"><h3>Bonus de la semaine</h3></div>
                <div className="bonus-grid">
                  <div className="bonus-box">
                    <div className="bonus-icon"><Icon name="refresh" /></div>
                    <div className="bonus-copy"><strong>Reroll</strong><small>Renouveler Lucky Boost et la vitrine</small>
                      <button className="secondary-button mini-buy" disabled={rerollUsed || balance < 150} onClick={buyReroll}>
                        {rerollUsed ? "Déjà utilisé" : <>Reroll · 150 <Icon name="gem" /></>}
                      </button>
                    </div>
                  </div>
                  <div className="bonus-box">
                    <div className="bonus-icon"><Icon name="gem" /></div>
                    <div className="bonus-copy"><strong>Carte fragments</strong><small>Une récompense gratuite cette semaine</small>
                      <button className="secondary-button mini-buy" disabled={fragmentClaimed} onClick={claimFragments}>{fragmentClaimed ? "Réclamée" : "Réclamer gratuitement"}</button>
                    </div>
                  </div>
                </div>
              </section>
          </section>

          <section className="shop-section" ref={sections.permanent}>
              <div className="shop-section-head"><h2>Permanents</h2><span className="section-note">Toujours disponibles</span></div>
              <div className="permanent-grid">
                {PACKS.filter((p) => ["memory", "anima", "rarity", "epico"].includes(p.id)).map((p) => (
                  <article className="permanent-tile" key={p.id}>
                    <Pack p={p} small />
                    <ShopBuy price={p.price} disabled={balance < p.price} added={addedId === p.id} onClick={() => buyPerm(p)} />
                  </article>
                ))}
              </div>
          </section>
        </main>

        <footer className="site-footer">
          <span>Bizave TCG · Prototype JSX indépendant · Aucun achat réel</span>
          <span>Conception basée sur les styles du dépôt <a className="footer-link" href="https://github.com/kabomane/bizave-tcg" target="_blank" rel="noreferrer">kabomane/bizave-tcg</a></span>
        </footer>
      </div>

      <nav className="mobile-nav" aria-label="Navigation mobile">
        {[["package", "Boosters"], ["layers", "Collection"], ["recycle", "Atelier"], ["users", "Amis"]].map(([ic, l], i) => (
          <a key={l} className={i === 0 ? "active" : undefined} href="#" onClick={(e) => e.preventDefault()}><Icon name={ic} /><span>{l}</span></a>
        ))}
      </nav>

      {ratesOpen && <RatesDialog groups={rateGroups} onClose={closeRates} />}

      {toastMsg && <div className="toast" role="status" aria-live="polite" key={toastMsg.k}>{toastMsg.m}</div>}
    </div>
  );
}
